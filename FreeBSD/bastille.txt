# 1. MACROS (Variables)
ext_if      = "em0"            # Host internet interface
jail_net    = "10.0.0.0/24"    # Isolated network for your jails
jailTomcat  = "10.0.0.80"      # Your specific Tomcat jail IP
tomcat_port = "8080"          # The internal port Tomcat listens on

# 2. TRANSLATION (NAT & Port Forwarding)
# NAT: Allows jailTomcat to send packets OUT to the internet (for pkg updates)
nat on $ext_if from $jail_net to any -> ($ext_if)

# Port Forward: Redirects incoming public port 8080 traffic to the internal jail
rdr on $ext_if proto tcp from any to any port 8080 -> $jailTomcat port $tomcat_port

# 3. DEFAULT DENY (Blocks all inbound and outbound traffic globally by default)
block all

# 4. EXPLICIT PASS RULES

# Internal Routing: Allow the host and jails to communicate freely inside local loops
pass quick on lo0 all
pass quick on bridge0 all

# --- HOST MACHINE PROTECTION ---
# SSH Access: Crucial rule to allow you to connect to the host via SSH
pass in quick on $ext_if proto tcp from any to any port 22 flags S/SA keep state

# Web Access: Allow the host to receive traffic on standard web port 80
pass in quick on $ext_if proto tcp from any to any port 80 flags S/SA keep state

# Host Outbound: Allow the host machine to reach the internet (DNS, updates, etc.)
pass out quick on $ext_if proto { tcp, udp, icmp } from ($ext_if) to any keep state


# --- JAIL TOMCAT STRICT RESTRICTIONS ---
# Inbound Traffic Rule: Allow ONLY port 8080 traffic to pass through the host to the jail
pass in quick on $ext_if proto tcp from any to $jailTomcat port $tomcat_port flags S/SA synproxy state

# Outbound Traffic Rule: Allow jailTomcat to establish connection OUTBOUND to the internet
# (Required for running pkg update and fetching Tomcat packages)
pass out quick on $ext_if proto { tcp, udp, icmp } from $jailTomcat to any keep state

This OpenJDK implementation may require procfs(5) mounted on /proc for some
applications.

If you have not done it yet, please do the following:

	mount -t procfs proc /proc

To make it permanent, you need the following lines in /etc/fstab:

	proc	/proc		procfs		rw	0	0

 >> $ sudo cat /usr/local/bastille/jails/jailTomcat/jail.conf
jailTomcat {
  enforce_statfs = 1;
  devfs_ruleset = 13;
  exec.clean;
  exec.consolelog = /var/log/bastille/jailTomcat_console.log;
  exec.start = '/bin/sh /etc/rc';
  exec.stop = '/bin/sh /etc/rc.shutdown';
  host.hostname = jailTomcat;
  mount.devfs;
  mount.fstab = /usr/local/bastille/jails/jailTomcat/fstab;
  path = /usr/local/bastille/jails/jailTomcat/root;
  securelevel = 2;
  osrelease = 15.1-RELEASE;

  vnet;
  vnet.interface = e0b_jailTomcat;
  exec.prestart += "epair0=\$(ifconfig epair create) && ifconfig \${epair0} up name e0a_jailTomcat && ifconfig \${epair0%a}b up name e0b_jailTomcat";
  exec.prestart += "ifconfig bridge0 addm e0a_jailTomcat";
  exec.prestart += "ifconfig e0a_jailTomcat ether 58:9c:fc:13:6f:7a";
  exec.prestart += "ifconfig e0b_jailTomcat ether 58:9c:fc:13:6f:7b";
  exec.prestart += "ifconfig e0a_jailTomcat description \"vnet0 host interface for Bastille jail jailTomcat\"";
  exec.poststop += "ifconfig e0a_jailTomcat destroy";

  mount.procfs;
}
